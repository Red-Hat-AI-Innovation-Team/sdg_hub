Quick Start

Load a flow, configure a model, generate synthetic data.

from sdg_hub import FlowRegistry, Flow

# Discover and load a flow
FlowRegistry.discover_flows()
path = FlowRegistry.get_flow_path("MCP Server Distillation")
flow = Flow.from_yaml(path)
# Configure model
flow.set_model_config(
    model="openai/gpt-4o",
    api_key="sk-...",
)

# Generate
result = flow.generate(dataset)
